FreeRADIUS has been installed.

1. Reaching the RADIUS service:
{{- if .Values.gateway.enabled }}
{{- $gwHost := default (include "st-common.names.fullname" .) (first .Values.gateway.hostnames) }}

  Routing is provided by the {{ .Values.gateway.implementation }} Gateway.
  Make sure DNS for "{{ $gwHost }}" resolves to the gateway's external address, then:

    radclient -x {{ $gwHost }}:{{ .Values.service.ports.auth }} auth <shared-secret>
{{- if .Values.tls.enabled }}
    radclient -x -P tcp -S <secret-file> {{ $gwHost }}:{{ .Values.service.ports.radsec }} auth <shared-secret>
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}

  export NODE_IP=$(kubectl get nodes --namespace {{ include "st-common.names.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  export AUTH_PORT=$(kubectl get --namespace {{ include "st-common.names.namespace" . }} -o jsonpath="{.spec.ports[?(@.name=='udp-auth')].nodePort}" services {{ include "st-common.names.fullname" . }})
  echo "RADIUS Auth: $NODE_IP:$AUTH_PORT (UDP)"
{{- else if contains "LoadBalancer" .Values.service.type }}

  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
        kubectl get --namespace {{ include "st-common.names.namespace" . }} svc -w {{ include "st-common.names.fullname" . }}

  export SERVICE_IP=$(kubectl get svc --namespace {{ include "st-common.names.namespace" . }} {{ include "st-common.names.fullname" . }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
  echo "RADIUS Auth: $SERVICE_IP:{{ .Values.service.ports.auth }} (UDP)"
{{- else if contains "ClusterIP" .Values.service.type }}

  export POD_NAME=$(kubectl get pods --namespace {{ include "st-common.names.namespace" . }} -l "app.kubernetes.io/name={{ .Chart.Name }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ include "st-common.names.namespace" . }} port-forward $POD_NAME {{ .Values.service.ports.auth }}:{{ .Values.containerPorts.auth }}
{{- end }}

   and persists across helm upgrades.
{{- end }}
